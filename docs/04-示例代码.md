## 基础示例

### 简单的 GET/SET 操作

```cpp
#include "galay-redis/async/RedisClient.h"
#include <galay-kernel/kernel/Runtime.h>
#include <iostream>

using namespace galay::redis;
using namespace galay::kernel;

Coroutine simpleExample(IOScheduler* scheduler)
{
    RedisClient client(scheduler);
    
    // 连接到 Redis
    auto conn = co_await client.connect("127.0.0.1", 6379);
    if (!conn) {
        std::cerr << "连接失败: " << conn.error().message() << std::endl;
        co_return;
    }
    std::cout << "连接成功" << std::endl;
    
    // SET 操作
    auto set_result = co_await client.set("mykey", "myvalue");
    if (set_result && set_result.value()) {
        std::cout << "SET 成功" << std::endl;
    }
    
    // GET 操作
    auto get_result = co_await client.get("mykey");
    if (get_result && get_result.value()) {
        auto& values = get_result.value().value();
        if (!values.empty() && values[0].isString()) {
            std::cout << "GET 结果: " << values[0].toString() << std::endl;
        }
    }
    
    co_await client.close();
}

int main()
{
    Runtime runtime;
    runtime.start();
    
    auto* scheduler = runtime.getNextIOScheduler();
    scheduler->spawn(simpleExample(scheduler));
    
    std::this_thread::sleep_for(std::chrono::seconds(5));
    runtime.stop();
    return 0;
}
```

### 带超时的操作

```cpp
Coroutine timeoutExample(IOScheduler* scheduler)
{
    RedisClient client(scheduler);
    
    auto conn = co_await client.connect("127.0.0.1", 6379)
                               .timeout(std::chrono::seconds(3));
    if (!conn) {
        std::cerr << "连接超时或失败" << std::endl;
        co_return;
    }
    
    // 设置 5 秒超时
    auto result = co_await client.get("mykey")
                                 .timeout(std::chrono::seconds(5));
    
    if (!result) {
        if (result.error().type() == REDIS_ERROR_TYPE_TIMEOUT_ERROR) {
            std::cerr << "操作超时" << std::endl;
        } else {
            std::cerr << "操作失败: " << result.error().message() << std::endl;
        }
        co_return;
    }
    
    // 处理结果
    if (result.value()) {
        auto& values = result.value().value();
        std::cout << "获取到 " << values.size() << " 个值" << std::endl;
    }
    
    co_await client.close();
}
```

## Pipeline 示例

### 批量操作

```cpp
Coroutine pipelineBatchExample(IOScheduler* scheduler)
{
    RedisClient client(scheduler);
    co_await client.connect("127.0.0.1", 6379);
    
    // 准备批量命令
    std::vector<std::vector<std::string>> commands;
    for (int i = 0; i < 100; ++i) {
        commands.push_back({
            "SET", 
            "key:" + std::to_string(i), 
            "value:" + std::to_string(i)
        });
    }
    
    // 执行 Pipeline
    auto start = std::chrono::steady_clock::now();
    auto result = co_await client.pipeline(commands)
                                 .timeout(std::chrono::seconds(10));
    auto end = std::chrono::steady_clock::now();
    
    if (result && result.value()) {
        auto duration = std::chrono::duration_cast<std::chrono::milliseconds>(end - start);
        std::cout << "Pipeline 完成 100 个操作，耗时: " 
                 << duration.count() << " ms" << std::endl;
    }
    
    co_await client.close();
}
```

### 基本连接池使用

```cpp
Coroutine poolBasicExample(IOScheduler* scheduler)
{
    // 创建连接池配置
    auto config = ConnectionPoolConfig::create("127.0.0.1", 6379, 2, 10);
    config.initial_connections = 2;
    config.acquire_timeout = std::chrono::seconds(5);
    
    RedisConnectionPool pool(scheduler, config);
    
    // 初始化连接池
    auto init = co_await pool.initialize().timeout(std::chrono::seconds(5));
    if (!init) {
        std::cerr << "连接池初始化失败" << std::endl;
        co_return;
    }
    
    // 获取连接
    auto conn_result = co_await pool.acquire().timeout(std::chrono::seconds(2));
    if (!conn_result) {
        std::cerr << "获取连接失败" << std::endl;
        pool.shutdown();
        co_return;
    }
    
    auto conn = conn_result.value();
    auto redis_client = conn->get();
    
    // 使用连接
    auto result = co_await redis_client->set("poolkey", "poolvalue");
    
    // 归还连接
    pool.release(conn);
    
    // 查看统计
    auto stats = pool.getStats();
    std::cout << "连接池统计:" << std::endl;
    std::cout << "  总连接: " << stats.total_connections << std::endl;
    std::cout << "  可用连接: " << stats.available_connections << std::endl;
    std::cout << "  活跃连接: " << stats.active_connections << std::endl;
    
    pool.shutdown();
}
```

```cpp
Coroutine poolWorker(IOScheduler* scheduler, RedisConnectionPool& pool, int id)
{
    for (int i = 0; i < 10; ++i) {
        auto conn_result = co_await pool.acquire();
        if (!conn_result) continue;
```

### 会话管理系统

```cpp
class SessionManager
{
public:
    SessionManager(IOScheduler* scheduler, RedisConnectionPool& pool)
        : m_scheduler(scheduler), m_pool(pool)
    {
    }
    
    Coroutine createSession(const std::string& user_id, const std::string& token)
    {
        auto conn_result = co_await m_pool.acquire();
        if (!conn_result) {
            co_return false;
        }
        
        ScopedConnection scoped(m_pool, conn_result.value());
        
        std::string key = "session:" + user_id;
        auto result = co_await scoped->setex(key, 3600, token)
                                     .timeout(std::chrono::seconds(2));
        
        co_return result.has_value();
    }
    
    Coroutine validateSession(const std::string& user_id)
    {
        auto conn_result = co_await m_pool.acquire();
        if (!conn_result) {
            co_return std::optional<std::string>();
        }
        
        ScopedConnection scoped(m_pool, conn_result.value());
        
        std::string key = "session:" + user_id;
        auto result = co_await scoped->get(key)
                                     .timeout(std::chrono::seconds(2));
        
        if (result && result.value()) {
            auto& values = result.value().value();
            if (!values.empty() && values[0].isString()) {
                co_return std::optional<std::string>(values[0].toString());
            }
        }
        
        co_return std::optional<std::string>();
    }
    
    Coroutine deleteSession(const std::string& user_id)
    {
        auto conn_result = co_await m_pool.acquire();
        if (!conn_result) {
            co_return false;
        }
        
        ScopedConnection scoped(m_pool, conn_result.value());
        
        std::string key = "session:" + user_id;
        auto result = co_await scoped->del(key)
                                     .timeout(std::chrono::seconds(2));
        
        co_return result.has_value();
    }

private:
    IOScheduler* m_scheduler;
    RedisConnectionPool& m_pool;
};

// 使用示例
Coroutine sessionExample(IOScheduler* scheduler)
{
    auto config = ConnectionPoolConfig::create("127.0.0.1", 6379, 5, 20);
    RedisConnectionPool pool(scheduler, config);
    co_await pool.initialize();
    
    SessionManager manager(scheduler, pool);
    
    // 创建会话
    bool created = co_await manager.createSession("user123", "token_abc");
    std::cout << "会话创建: " << (created ? "成功" : "失败") << std::endl;
    
    // 验证会话
    auto token = co_await manager.validateSession("user123");
    if (token.has_value()) {
        std::cout << "会话有效，token: " << token.value() << std::endl;
    }
    
    // 删除会话
    bool deleted = co_await manager.deleteSession("user123");
    std::cout << "会话删除: " << (deleted ? "成功" : "失败") << std::endl;
    
    pool.shutdown();
}
```

### 分布式计数器

```cpp
class DistributedCounter
{
public:
    DistributedCounter(RedisClient& client, const std::string& key)
        : m_client(client), m_key(key)
    {
    }
    
    Coroutine increment(int64_t delta = 1)
    {
        auto result = co_await m_client.execute("INCRBY", {m_key, std::to_string(delta)})
                                       .timeout(std::chrono::seconds(2));
        
        if (result && result.value()) {
            auto& values = result.value().value();
            if (!values.empty() && values[0].isInteger()) {
                co_return std::optional<int64_t>(values[0].toInteger());
            }
        }
        
        co_return std::optional<int64_t>();
    }
    
    Coroutine decrement(int64_t delta = 1)
    {
        auto result = co_await m_client.execute("DECRBY", {m_key, std::to_string(delta)})
                                       .timeout(std::chrono::seconds(2));
        
        if (result && result.value()) {
            auto& values = result.value().value();
            if (!values.empty() && values[0].isInteger()) {
                co_return std::optional<int64_t>(values[0].toInteger());
            }
        }
        
        co_return std::optional<int64_t>();
    }
    
    Coroutine get()
    {
        auto result = co_await m_client.get(m_key)
                                       .timeout(std::chrono::seconds(2));
        
        if (result && result.value()) {
            auto& values = result.value().value();
            if (!values.empty() && values[0].isString()) {
                co_return std::optional<int64_t>(std::stoll(values[0].toString()));
            }
        }
        
        co_return std::optional<int64_t>();
    }

private:
    RedisClient& m_client;
    std::string m_key;
};
```
